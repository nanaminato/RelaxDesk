cmake_minimum_required(VERSION 3.24)
project(RelaxDeskNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(RelaxDesk.Native SHARED graphics.cpp codec.cpp)
target_compile_definitions(RelaxDesk.Native PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(RelaxDesk.Native PRIVATE /W4 /EHsc)
target_link_libraries(RelaxDesk.Native PRIVATE d3d11 dxgi mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32)
add_executable(RelaxDesk.NativeProbe probe.cpp)
target_compile_definitions(RelaxDesk.NativeProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(RelaxDesk.NativeProbe PRIVATE RelaxDesk.Native user32 gdi32)
